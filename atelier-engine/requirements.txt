# The atelier engine's dependencies.
#
# Every pin is a version that had been on the index for at least thirty days
# when it was chosen, which is the same rule pnpm enforces on the site
# (`minimumReleaseAge: 43200`). pip has no equivalent setting, so the check is
# done by hand and the result written here: the date beside each pin is when
# that version was published, and all of them were checked on 2026-10-09.
#
# Keep this list short. A model that `transformers` can load does not also need
# its own package — that is one supply chain instead of two, which is why there
# is no `sam2` here even though SAM 2.1 is used.

# cu130 because the card is an Ada 4060 and the installed driver (592.82)
# already runs CUDA 13 — the plain PyPI wheel for Windows is CPU-only, which
# would silently work and take a thousand times as long.
--extra-index-url https://download.pytorch.org/whl/cu130

torch==2.14.0+cu130      # published 2026-09-02

# Not a choice: `transformers` loads SAM 2.1 through Sam2ImageProcessor, which
# imports torchvision and fails without it. Same publisher as torch and
# published the same day, which is the pair that is meant to go together — 0.29.1
# exists but was eight days old, so it is held back.
torchvision==0.29.0+cu130  # published 2026-09-02

transformers==5.16.1     # published 2026-08-26
pillow==12.3.0           # published 2026-07-01

# Held at 2.4.x deliberately: numpy 2.5 requires Python 3.12, and 3.11 is what
# is installed on this machine. Nothing here needs 2.5, so the engine stays on
# 3.11 rather than the whole toolchain moving for one pin.
numpy==2.4.6             # published 2026-05-18

# ---- the service ----
fastapi==0.141.1         # published 2026-07-29
uvicorn==0.52.4          # published 2026-08-19
python-multipart==0.0.32 # published 2026-06-04

# Pinned although nothing here imports them. Pinning only what you import
# constrains none of what it drags in: asking for fastapi alone resolved
# pydantic and pydantic-core published *the previous day*, and starlette at
# fifteen days. pnpm applies `minimumReleaseAge` to the whole tree; pip applies
# nothing to anything, so the tree is held here by hand.
#
# Both are the newest that are old enough and still satisfy fastapi 0.141.1,
# which wants starlette>=0.46.0 and pydantic>=2.9.0. pydantic-core is left to
# pydantic, which pins it exactly and to a build of the same age.
starlette==1.6.0         # published 2026-08-08
pydantic==2.13.5         # published 2026-08-28

# ---- filling in behind a cut ----
#
# LaMa, as ONNX rather than as a TorchScript file or the original checkpoint.
# Three reasons, in order: an ONNX graph is data and cannot execute code on
# load, where TorchScript and pickle can, and the well-known big-lama.pt files
# are individuals' GitHub releases; the original checkpoint needs the whole lama
# codebase to read it; and this runs on the CPU, so filling in competes for none
# of the 8 GB the segmenter needs.
#
# 1.31.0 was published today and 1.30 within the month, so both are held back.
onnxruntime==1.29.0      # published 2026-08-17

# ---- image + text -> image ----
#
# Qwen-Image-Edit, chosen over FLUX.1 schnell because it follows an instruction
# ("make the shirt red") rather than merely restyling. It is 20B and the honest
# cost is quality: on an 8 GB card only the smallest quantisations fit, and the
# text encoder and VAE have to live in system RAM.
#
# `gguf` is what lets diffusers read a quantised checkpoint; `accelerate` is the
# CPU-offload machinery that makes a model larger than the card run at all.
diffusers==0.40.0        # published 2026-08-20
gguf==0.19.0             # published 2026-05-06
accelerate==1.14.0       # published 2026-06-11
